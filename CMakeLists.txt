cmake_minimum_required(VERSION 3.13)
project(sharp_ax8860 CXX)

# AX8860 BSP (headers + libs of ax_engine / ax_sys / ax_vdec), e.g. .../arm64_glibc
set(BSP_DIR "" CACHE PATH "AX8860 BSP directory containing include/ and lib/")
option(SHARP_POST_ONLY "Build a host (x86) binary that only runs --post-only" OFF)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_FLAGS_RELEASE "-O3")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Threads REQUIRED)

function(sharp_setup t)
  target_include_directories(${t} PRIVATE src third_party)
  # 热路径全用单精度 (A55 上双精度超越函数很慢)
  target_compile_options(${t} PRIVATE -fsingle-precision-constant -ffast-math)
  target_link_libraries(${t} PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
  if(SHARP_POST_ONLY)
    target_compile_definitions(${t} PRIVATE SHARP_POST_ONLY)
  else()
    target_include_directories(${t} PRIVATE ${BSP_DIR}/include)
    target_link_directories(${t} PRIVATE ${BSP_DIR}/lib)
    # libmvec: glibc >= 2.38 的 NEON expf/logf (见 sharp_post.hpp)
    target_link_libraries(${t} PRIVATE ax_engine ax_vdec ax_sys mvec)
    target_link_options(${t} PRIVATE -Wl,-rpath-link,${BSP_DIR}/lib)
  endif()
endfunction()

if(NOT SHARP_POST_ONLY AND NOT BSP_DIR)
  message(FATAL_ERROR "set -DBSP_DIR=<AX8860 BSP dir with include/ and lib/>")
endif()

# 命令行工具
add_executable(sharp_ax src/sharp_ax.cpp)
sharp_setup(sharp_ax)
install(TARGETS sharp_ax RUNTIME DESTINATION bin)

# C 接口动态库 libsharp_ax.so (Python ctypes 用)
if(NOT SHARP_POST_ONLY)
  add_library(sharp_ax_shared SHARED src/sharp_capi.cpp)
  set_target_properties(sharp_ax_shared PROPERTIES OUTPUT_NAME sharp_ax)
  sharp_setup(sharp_ax_shared)
  install(TARGETS sharp_ax_shared LIBRARY DESTINATION lib)
  install(FILES src/sharp_ax.h DESTINATION include)
endif()
